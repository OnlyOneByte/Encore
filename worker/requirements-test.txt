# Encore worker — the TOOLING the four worker gates run on, pinned like the runtime set. Separate from
# requirements.txt so running the gates costs no multi-GB torch download: the suite fakes every heavy seam.
#
#     pip install -r requirements-test.txt     # run the gates
#     pip install -r requirements.txt          # additionally run real jobs
#
# pytest 8.4.2 is why pytest.ini deliberately leaves `faulthandler_exit_on_timeout` unset: on 8.4.2 it
# is an unknown key that warns rather than errors. Bumping to pytest 9 means revisiting that decision.
pytest==8.4.2
# asyncio_mode=auto in pytest.ini is this plugin's setting — without it every async test function is
# silently skipped as an un-awaited coroutine rather than failing.
pytest-asyncio==1.4.0
# Backs pytest.ini's `timeout` / `timeout_method`: a wedged test fails alone instead of erasing the
# rest of the run's signal.
pytest-timeout==2.4.0
# ruff is lint AND formatter (see pyproject.toml). EXACT, not `>=`: a formatter's output is part of the
# gate, so a newer minor can turn `ruff format --check` red on files nobody touched.
ruff==0.16.8
# mypy's own config (python_version, files) is in pyproject.toml. `python_version = "3.12"` is the
# TARGET, not the interpreter this must run on — mypy on 3.10 checks 3.12 semantics fine.
mypy==2.3.1
